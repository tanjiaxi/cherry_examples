```go
// 客户端统计
type ClientMetrics struct {
    RequestsSent     int64  // 发送的请求数
    ResponsesReceived int64 // 收到的响应数

    ReceiveQPS       float64
}

// 服务端统计 (需要在框架中添加)
type ServerMetrics struct {

```

**服务端添加统计代码**:
```go
// 在 Actor 消息处理中添加
var (
    serverRequestCount  int64
    serverResponseCount int64
)

func (a *Actor) processMessage(msg *Message) {
    atomic.AddInt64(&serverRequestCount, 1)
    // ... 处理逻辑
    atomic.AddInt64(&serverResponseCount, 1)
}

// 定时输出
func PrintServerMetrics() {
    // 每秒输出一次服务端 QPS
}
```

### 3.3 QPS 差异分析

```
如果 客户端发送QPS > 服务端处理QPS:
    → 网络拥塞 或 服务端处理能力不足

如果 客户端发送QPS ≈ 服务端处理QPS，但延迟高:
    → 单个请求处理慢，需要分析具体接口

如果 服务端处理QPS > 客户端接收QPS:
    → 客户端处理响应慢 或 网络回程拥塞
```

## 第四阶段：瓶颈定位

### 4.1 瓶颈类型判断流程

```
                    ┌─────────────────┐
                    │  延迟是否正常?   │
                    └────────┬────────┘
                             │
              ┌──────────────┴──────────────┐
              │                             │
         延迟正常                        延迟高
              │                             │
              ▼                             ▼
    ┌─────────────────┐           ┌─────────────────┐
    │ QPS 是否达到预期? │           │ 哪个阶段延迟高?  │
    └────────┬────────┘           └────────┬────────┘
             │                             │
    ┌────────┴────────┐         ┌──────────┼──────────┐
    │                 │         │          │          │
  QPS正常          QPS低      连接慢     请求慢     响应慢
    │                 │         │          │          │
    ▼                 ▼         ▼          ▼          ▼
  测试完成      检查客户端    检查网络   检查服务端  检查网络
              是否有瓶颈    /服务端    处理逻辑   回程
```

### 4.2 各类瓶颈的特征

| 瓶颈类型 | 特征 | 验证方法 |
|----------|------|----------|
| **CPU 瓶颈** | CPU 使用率接近 100% | top/htop 观察 |
| **内存瓶颈** | 内存持续增长，GC 频繁 | pprof heap |
| **网络瓶颈** | 带宽打满，延迟抖动大 | iftop/nethogs |
| **数据库瓶颈** | 数据库 CPU 高，连接池满 | pg_stat_activity |
| **框架瓶颈** | Goroutine 数量暴涨，Actor 队列积压 | pprof goroutine |
| **锁竞争** | CPU 不高但 QPS 上不去 | pprof mutex |

### 4.3 框架特定瓶颈检查

**Actor 消息队列监控**:

```bash
# 1. 采集 30 秒 CPU profile
go tool pprof http://10.10.10.251:6060/debug/pprof/profile?seconds=30
```


```
最大连接数 = (核心数 * 2) + 有效磁盘数

```

## 附录：服务端监控代码实现

### B.1 服务端 QPS 统计器

```go
// internal/metrics/server_metrics.go
package metrics

import (
    "sync"
    "sync/atomic"
    "time"
)

type ServerMetrics struct {
    // 请求计数
    requestCount  int64
    responseCount int64
    errorCount    int64
    
    // 延迟统计
    totalLatency int64
    maxLatency   int64
    
    // 滑动窗口
    windowMu     sync.Mutex
    windowQPS    []float64
    windowTime   []time.Time
    windowSize   int
    
    // 每秒统计
    lastSecond   time.Time
    lastSecCount int64
}

var (
    routeMetrics = make(map[string]*ServerMetrics)
    metricsMu    sync.RWMutex
)

func GetRouteMetrics(route string) *ServerMetrics {
    metricsMu.RLock()
    m := routeMetrics[route]
    metricsMu.RUnlock()
    
    if m == nil {
        metricsMu.Lock()
        m = &ServerMetrics{windowSize: 60}
        routeMetrics[route] = m
        metricsMu.Unlock()
    }
    return m
}

func (m *ServerMetrics) RecordRequest() {
    atomic.AddInt64(&m.requestCount, 1)
}

func (m *ServerMetrics) RecordResponse(latencyMs int64, isError bool) {
    atomic.AddInt64(&m.responseCount, 1)
    atomic.AddInt64(&m.totalLatency, latencyMs)
    
    if isError {
        atomic.AddInt64(&m.errorCount, 1)
    }
    
    // 更新最大延迟
    for {
        cur := atomic.LoadInt64(&m.maxLatency)
        if latencyMs <= cur || atomic.CompareAndSwapInt64(&m.maxLatency, cur, latencyMs) {
            break
        }
    }
}

func (m *ServerMetrics) GetStats() (requests, responses, errors, avgLatency, maxLatency int64) {
    requests = atomic.LoadInt64(&m.requestCount)
    responses = atomic.LoadInt64(&m.responseCount)
    errors = atomic.LoadInt64(&m.errorCount)
    maxLatency = atomic.LoadInt64(&m.maxLatency)
    
    if responses > 0 {
        avgLatency = atomic.LoadInt64(&m.totalLatency) / responses
    }
    return
}

// PrintAllMetrics 打印所有路由的统计
func PrintAllMetrics() {
    metricsMu.RLock()
    defer metricsMu.RUnlock()
    
    for route, m := range routeMetrics {
        req, resp, errs, avg, max := m.GetStats()
        if req > 0 {
            log.Infof("[ServerMetrics] %s: Req=%d Resp=%d Err=%d Avg=%dms Max=%dms",
                route, req, resp, errs, avg, max)
        }
    }
}
```

### B.3 延迟百分位统计

```go
// internal/metrics/latency_histogram.go
package metrics

import (
    "sort"
    "sync"
)

type LatencyHistogram struct {
    mu      sync.Mutex
    samples []int64
    maxSize int
}

func NewLatencyHistogram(maxSize int) *LatencyHistogram {
    return &LatencyHistogram{
        samples: make([]int64, 0, maxSize),
        maxSize: maxSize,
    }
}

func (h *LatencyHistogram) Record(latencyMs int64) {
    h.mu.Lock()
    defer h.mu.Unlock()
    
    if len(h.samples) >= h.maxSize {
        // 移除最旧的样本
        h.samples = h.samples[1:]
    }
    h.samples = append(h.samples, latencyMs)
}

func (h *LatencyHistogram) Percentile(p float64) int64 {
    h.mu.Lock()
    defer h.mu.Unlock()
    
    if len(h.samples) == 0 {
        return 0
    }
    
    sorted := make([]int64, len(h.samples))
    copy(sorted, h.samples)
    sort.Slice(sorted, func(i, j int) bool { return sorted[i] < sorted[j] })
    
    idx := int(float64(len(sorted)-1) * p)
    return sorted[idx]
}

func (h *LatencyHistogram) GetPercentiles() (p50, p90, p95, p99, max int64) {
    h.mu.Lock()
    defer h.mu.Unlock()
    
    if len(h.samples) == 0 {
        return
    }
    
    sorted := make([]int64, len(h.samples))
    copy(sorted, h.samples)
    sort.Slice(sorted, func(i, j int) bool { return sorted[i] < sorted[j] })
    
    n := len(sorted)
    p50 = sorted[int(float64(n-1)*0.50)]
    p90 = sorted[int(float64(n-1)*0.90)]
    p95 = sorted[int(float64(n-1)*0.95)]
    p99 = sorted[int(float64(n-1)*0.99)]
    max = sorted[n-1]
    return
}
```

```go
func TestConnectionOnly(count int, batchSize int, interval time.Duration) {
    var wg sync.WaitGroup
    var successCount, failCount int64
    
    startTime := time.Now()
    
    for i := 0; i < count; i += batchSize {
        batch := min(batchSize, count-i)
        
        for j := 0; j < batch; j++ {
            wg.Add(1)
            go func(idx int) {
                defer wg.Done()
                
                connStart := time.Now()
                robot := robotclient.New(pomeloClient.New())
                
                err := robot.ConnectToWebSocket(gateAddr)
                connLatency := time.Since(connStart)
                
                if err != nil {
                    atomic.AddInt64(&failCount, 1)
                    log.Errorf("Connect failed: %v", err)
                } else {
                    atomic.AddInt64(&successCount, 1)
                    recordMetrics("Connect", connLatency, false)
                    
                    // 保持连接，不断开
                    connectedRobots = append(connectedRobots, robot)
                }
            }(i + j)
        }
        
        wg.Wait()
        
        if interval > 0 {
            time.Sleep(interval)
        }
        
        log.Infof("Progress: %d/%d, Success: %d, Fail: %d",
            i+batch, count, successCount, failCount)
    }
    
    totalTime := time.Since(startTime)
    log.Infof("Connection test done: %d connections in %v (%.1f/s)",
        successCount, totalTime, float64(successCount)/totalTime.Seconds())
}
```

```go
func RunSteppedLoadTest(steps []LoadStep) []StepResult {
    var results []StepResult

        log.Infof("=== Step: %d connections, %v duration ===", 
            step.Connections, step.Duration)

    }
    
    return results
}

func rampUpConnections(target int, duration time.Duration) {
    current := len(connectedRobots)
    toAdd := target - current
    
    if toAdd <= 0 {
        return
    }
    
    batchSize := 50
    interval := duration / time.Duration(toAdd/batchSize+1)
    
    for i := 0; i < toAdd; i += batchSize {
        batch := min(batchSize, toAdd-i)
        addConnections(batch)
        time.Sleep(interval)
    }
}
```
