```go
// internal/component/metrics/business_metrics.go
package metrics

import (
    "github.com/prometheus/client_golang/prometheus"
    "github.com/prometheus/client_golang/prometheus/promauto"
)

var (
    // 在线人数
    OnlineUsers = promauto.NewGauge(prometheus.GaugeOpts{
        Name: "game_online_users",
        Help: "Current online users",
    })
    
    // Spin次数
    SpinTotal = promauto.NewCounterVec(
        prometheus.CounterOpts{
            Name: "game_spin_total",
            Help: "Total number of spins",
        },
        []string{"machine_id", "result"},
    )
    
    // 下注金额
    BetAmount = promauto.NewHistogram(prometheus.HistogramOpts{
        Name:    "game_bet_amount",
        Help:    "Bet amount distribution",
        Buckets: []float64{10, 50, 100, 500, 1000, 5000, 10000},
    })
    
    // 响应时间
    ResponseTime = promauto.NewHistogramVec(
        prometheus.HistogramOpts{
            Name:    "game_response_time_seconds",
            Help:    "Response time in seconds",
            Buckets: prometheus.DefBuckets,
        },
        []string{"route"},
    )
    
    // RTP
    RTP = promauto.NewGaugeVec(
        prometheus.GaugeOpts{
            Name: "game_rtp",
            Help: "Return to player percentage",
        },
        []string{"machine_id"},
    )
)

// 使用示例
func (a *ActorPlayer) handleSpin(req *SpinRequest) {
    start := time.Now()
    defer func() {
        // 记录响应时间
        ResponseTime.WithLabelValues("player.spin").
            Observe(time.Since(start).Seconds())
    }()
    
    // 记录下注
    BetAmount.Observe(float64(req.BetAmount))
    
    // 执行Spin
    result := a.doSpin(req)
    
    // 记录结果
    SpinTotal.WithLabelValues(
        fmt.Sprint(a.MachineId),
        result.Type,  // "win" or "lose"
    ).Inc()
    
    // 更新RTP
    rtp := float64(a.TotalWin) / float64(a.TotalBet)
    RTP.WithLabelValues(fmt.Sprint(a.MachineId)).Set(rtp)
}
```

```go
// 1. 请求签名验证
func verifySignature(req *http.Request) bool {
    timestamp := req.Header.Get("X-Timestamp")
    signature := req.Header.Get("X-Signature")
    
    // 检查时间戳(防重放攻击)
    ts, _ := strconv.ParseInt(timestamp, 10, 64)
    if time.Now().Unix() - ts > 300 {
        return false  // 超过5分钟
    }
    
    // 验证签名
    data := req.URL.Path + timestamp + req.Body
    expected := hmac.SHA256(data, secretKey)
    return signature == expected
}

// 2. 频率限制
type RateLimiter struct {
    limiters map[int64]*rate.Limiter
    mu       sync.RWMutex
}

func (r *RateLimiter) Allow(userId int64) bool {
    r.mu.RLock()
    limiter, exists := r.limiters[userId]
    r.mu.RUnlock()
    
    if !exists {
        r.mu.Lock()
        limiter = rate.NewLimiter(2, 5)  // 每秒2次,突发5次
        r.limiters[userId] = limiter
        r.mu.Unlock()
    }
    
    return limiter.Allow()
}

// 3. 行为分析
type BehaviorAnalyzer struct {
    patterns map[int64]*PlayerPattern
}

type PlayerPattern struct {
    SpinCount     int64
    WinRate       float64
    AvgBetAmount  int64
    Suspicious    bool
}

func (a *BehaviorAnalyzer) Analyze(userId int64) {
    pattern := a.patterns[userId]

    }
```

```go
func encryptSensitiveData(data string) (string, error) {
    block, err := aes.NewCipher(encryptionKey)
    if err != nil {
        return "", err
    }
    
    gcm, err := cipher.NewGCM(block)
    if err != nil {
        return "", err
    }
    
    nonce := make([]byte, gcm.NonceSize())
    rand.Read(nonce)
    
    ciphertext := gcm.Seal(nonce, nonce, []byte(data), nil)
    return base64.StdEncoding.EncodeToString(ciphertext), nil
}
```

```go
// 执行GM命令
func executeGMCommand(admin string, cmd string, args []string) (string, error) {
    command, exists := gmCommands[cmd]
    if !exists {
        return "", errors.New("command not found")
    }
    
    // 权限检查
    if !hasPermission(admin, command.Permission) {
        return "", errors.New("permission denied")
    }
    
    // 记录审计日志
    auditLogger.Info("gm command",
        zap.String("admin", admin),
        zap.String("command", cmd),
        zap.Strings("args", args),
    )
    
    // 执行命令
    return command.Handler(args)
}
```

### 5.2 数据分析平台

```sql
-- 关键指标SQL

-- 1. DAU(日活跃用户)
SELECT DATE(login_time) as date, COUNT(DISTINCT user_id) as dau
FROM login_log
WHERE login_time >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY DATE(login_time)
ORDER BY date;

-- 2. 新增用户
SELECT DATE(create_time) as date, COUNT(*) as new_users
FROM players
WHERE create_time >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY DATE(create_time);

-- 3. 留存率
WITH cohort AS (
    SELECT user_id, DATE(create_time) as cohort_date
    FROM players
)
SELECT 
    c.cohort_date,
    COUNT(DISTINCT c.user_id) as cohort_size,
    COUNT(DISTINCT CASE WHEN DATE(l.login_time) = c.cohort_date + 1 THEN l.user_id END) * 100.0 / COUNT(DISTINCT c.user_id) as day1_retention,
    COUNT(DISTINCT CASE WHEN DATE(l.login_time) = c.cohort_date + 7 THEN l.user_id END) * 100.0 / COUNT(DISTINCT c.user_id) as day7_retention
FROM cohort c
LEFT JOIN login_log l ON c.user_id = l.user_id
GROUP BY c.cohort_date
ORDER BY c.cohort_date DESC;

-- 4. ARPU(平均每用户收入)
SELECT 
    DATE(recharge_time) as date,
    SUM(amount) / COUNT(DISTINCT user_id) as arpu
FROM recharge_log
WHERE recharge_time >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY DATE(recharge_time);

-- 5. 付费率
SELECT 
    DATE(login_time) as date,
    COUNT(DISTINCT CASE WHEN r.user_id IS NOT NULL THEN l.user_id END) * 100.0 / COUNT(DISTINCT l.user_id) as pay_rate
FROM login_log l
LEFT JOIN recharge_log r ON l.user_id = r.user_id AND DATE(l.login_time) = DATE(r.recharge_time)
WHERE l.login_time >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY DATE(login_time);
```
